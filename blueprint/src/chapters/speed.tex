\chapter{Directional speed limits}
\label{chap:speed}

Along a unit direction $u$, the quadrotor's velocity component is driven by
$F\inner{b_3}{u}-g\inner{\up}{u}-c_D\norm{v}\inner{v}{u}$, with the collective
thrust $F\in[0,4f_{\mathrm{motor}}/m]$ and $b_3$ inside the tilt envelope. No
artificial speed cap is involved: the limit comes from motors, tilt and drag, and
it depends on the direction.

\begin{definition}[Reach, push, directional speed]
\label{def:dir-speed}
\lean{VisionTracking.QuadParams.maxAcc, VisionTracking.QuadParams.reach,
  VisionTracking.QuadParams.push, VisionTracking.QuadParams.dirSpeed}
\leanok
\uses{def:quad-params}
$\mathrm{reach}(u)=\cos\max(0,\ang(u,\up)-\theta_{\max})$;
$\mathrm{push}(u)=F_{\max}\max(0,\mathrm{reach}(u))-g\inner{\up}{u}$;
$\mathrm{dirSpeed}(u)=\sqrt{\max(0,\mathrm{push}(u))/c_D}$. Straight down this is
$\sqrt{g/c_D}$, the terminal fall speed. Horizontally it is
$\sqrt{F_{\max}\sin\theta_{\max}/c_D}$, and straight up
$\sqrt{(F_{\max}-g)/c_D}$.
\end{definition}

\begin{lemma}[Tilt envelope bounds reach]
\label{lem:reach}
\lean{VisionTracking.inner_le_reach, VisionTracking.QuadTraj.thrust_le,
  VisionTracking.dir_step_le}
\leanok
\uses{def:dir-speed, def:quad-traj}
$\inner{b_3}{u}\le\mathrm{reach}(u)$ whenever $\ang(b_3,\up)\le\theta_{\max}$. Hence
$\inner{v_{k+1}}{u}\le\inner{v_k}{u}+\delta(\mathrm{push}(u)-c_D\norm{v_k}\inner{v_k}{u})$.
\end{lemma}

\begin{proof}
\leanok
Use the triangle inequality for angles and the fact that cosine decreases on
$[0,\pi]$.
\end{proof}

\begin{lemma}[Excess contracts]
\label{lem:dir-excess}
\lean{VisionTracking.dir_excess_step, VisionTracking.dir_displacement_le}
\leanok
\uses{lem:reach}
Suppose $\delta c_D v_{\mathrm{env}}\le1/2$ and $\mathrm{push}(u)\le c_DV^2$ with
$0<V\le v_{\mathrm{env}}$. Then $\inner{v}{u}-V$ contracts by $1-2\delta c_DV$ per
step once positive, and
$\inner{x_k-x_0}{u}\le k\delta V+\max(0,\inner{v_0}{u}-V)/(2c_DV)$.
\end{lemma}

\begin{proof}
\leanok
Split on the sign of $\inner{v_k}{u}$ and on its position relative to $V$; above
$V$, quadratic drag beats the push. Then telescope.
\end{proof}

\begin{theorem}[Directional escape]
\label{thm:speed}
\lean{VisionTracking.not_maintainable_of_dirSpeed_lt,
  VisionTracking.straightTarget_escapes_of_displacement, VisionTracking.straightTarget,
  VisionTracking.straightTarget_startsAt, VisionTracking.inner_displacement_le,
  VisionTracking.straightTarget_escapes}
\leanok
\uses{lem:dir-excess, def:maintainable}
If the target may fly along $u$ (within its climb limit) faster than
$\mathrm{dirSpeed}(u)$, no full-information strategy keeps it in view, whatever the
motors, attitude and camera do. Straight down, this is the \emph{dive limit}: the
quadrotor cannot follow a target sinking faster than $\sqrt{g/c_D}$.
\end{theorem}

\begin{proof}
\leanok
Fly the target straight along $u$ at $v_{\max}$. Its advance outruns the bound of
Lemma~\ref{lem:dir-excess}.
\end{proof}
